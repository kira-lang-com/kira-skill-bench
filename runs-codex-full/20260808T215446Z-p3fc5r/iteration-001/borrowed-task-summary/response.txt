struct FocusTask {
    let minutes: Int;
    let weight: Int;
}

function weightedMinutes(tasks: borrow [FocusTask]) -> Int {
    var total = 0;
    for task in tasks {
        total = total + task.minutes * task.weight;
    }
    return total;
}

function longTaskCount(tasks: borrow [FocusTask], threshold: Int) -> Int {
    var count = 0;
    for task in tasks {
        if task.minutes >= threshold {
            count = count + 1;
        }
    }
    return count;
}

@Main
function main() {
    let tasks: [FocusTask] = [
        FocusTask { minutes: 25, weight: 2 },
        FocusTask { minutes: 60, weight: 1 }
    ];
    print(weightedMinutes(tasks));
    print(longTaskCount(tasks, 30));
    return
}
