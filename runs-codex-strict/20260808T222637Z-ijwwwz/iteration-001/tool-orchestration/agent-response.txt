import KiraTools

function totalIssuePoints() -> Int {
    let tools = makeKiraTools()
    let issues = tools.github.searchIssues("kira-lang/kira", "LLVM")
    var total: Int = 0

    for issue in issues {
        let detail = tools.github.getIssue(issue.id)
        total = total + detail.points
    }

    return total
}

@Main function main() {
    print(totalIssuePoints())
    return
}
